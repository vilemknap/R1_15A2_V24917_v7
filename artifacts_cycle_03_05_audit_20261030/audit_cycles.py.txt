   'model_observed_ratio_before':b['cal_metrics']['model_obs_ratio'],'model_observed_ratio_after':a['cal_metrics']['model_obs_ratio'],
   'geh_ge20_before':b['cal_metrics']['geh_ge20'],'geh_ge20_after':a['cal_metrics']['geh_ge20'],

   'gap':g['gap'],'rows_reported':g['rows'],'ls_search_incomplete':g['ls_search_incomplete'],
   'attempts':[{'trial':r['trial'],'scale':r['candidate']['scale'],'status':r['status'],
                'geh':r['decision']['cal_objective_after'],'reasons':r['decision']['reasons']} for r in g['attempts']],
   'longguard':{k:guard[k] for k in ('total_volume','below_min_km','support_clean','fine_100_200_rmse','fine_cdf_verified','long_generation_permitted')}},

 'td_evaluations':60,'budget':td['progressive_budget'],'mode':j['mode'],'native_gap':td['gap'],
 'solver_stop_reason':j['stop_reason'],'outer_incomplete_reason':td['inner']['stop_reason'],'complete_coordinate_polls':j['polls_completed'],

 'supply_cycle_4':{'status':supply['status'],'elapsed_seconds':supply['elapsed_s'],'geh_before':parent['raw_cal'],
